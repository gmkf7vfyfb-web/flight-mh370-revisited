      """Dependency-free, public-data B777-200ER fuel/performance approximation.

      This is a clean-room audit model, not a transcription of the Ulich workbook and
      not a substitute for Boeing or Rolls-Royce performance data.  Its intended use
      is to expose what can and cannot be reproduced from redistributable inputs.

      All external values and model choices are catalogued in
      ``data/model_parameters.json``.  In particular, this module contains no fuel
      quantity at 18:22 UTC, no 0.45 tonne dispersion, and no exhaustion-time target.
      """

      from __future__ import annotations

      from dataclasses import asdict, dataclass
      from datetime import datetime, timezone
      import math
      from typing import Iterable, Sequence


      G0_M_S2 = 9.80665
      GAS_CONSTANT_AIR_J_KG_K = 287.05287
      HEAT_CAPACITY_RATIO = 1.4
      SEA_LEVEL_PRESSURE_PA = 101_325.0
      SEA_LEVEL_TEMPERATURE_K = 288.15
      TROPOPAUSE_TEMPERATURE_K = 216.65
      TROPOPAUSE_HEIGHT_M = 11_000.0
      LAPSE_RATE_K_M = -0.0065
      FT_TO_M = 0.3048
      FPM_TO_M_S = 0.00508
      KT_TO_M_S = 0.514444


      @dataclass(frozen=True)
      class ModelParameters:
          """Executable parameters; provenance is stored beside them in JSON."""

          wing_area_m2: float = 427.8
          clean_cd0: float = 0.024
          induced_drag_k: float = 0.047
          engine_count: int = 2
          max_static_thrust_per_engine_n: float = 411_480.0
          fuel_flow_scale: float = 1.0

          # ICAO certification power fraction, uninstalled fuel flow (kg/s/engine),
          # and BFFM2 installation factor.  Piecewise-linear interpolation between
          # these points is an explicit conditional approximation.
          lto_points: tuple[tuple[float, float, float], ...] = (
              (0.07, 1.00, 1.100),
              (0.30, 3.10, 1.020),
              (0.85, 3.91, 1.013),
              (1.00, 5.33, 1.010),
          )

          def validate(self) -> None:
              values = (
                  self.wing_area_m2,
                  self.clean_cd0,
                  self.induced_drag_k,
                  self.max_static_thrust_per_engine_n,
                  self.fuel_flow_scale,
              )
              if not all(math.isfinite(value) and value > 0 for value in values):
                  raise ValueError("model parameters must be finite and positive")
              if self.engine_count != 2:
                  raise ValueError("this audit fixture is specific to a twin-engine 777")
              if len(self.lto_points) < 2:

      def isa_temperature_k(pressure_altitude_ft: float) -> float:
          """Return ISA temperature for a geopotential pressure altitude."""

          height_m = pressure_altitude_ft * FT_TO_M
          return max(
              SEA_LEVEL_TEMPERATURE_K + LAPSE_RATE_K_M * height_m,
              TROPOPAUSE_TEMPERATURE_K,
          )


      def pressure_from_pressure_altitude_pa(pressure_altitude_ft: float) -> float:
          """Return ISA pressure represented by a pressure-altitude indication."""

          height_m = pressure_altitude_ft * FT_TO_M
          if height_m <= TROPOPAUSE_HEIGHT_M:
              temperature = SEA_LEVEL_TEMPERATURE_K + LAPSE_RATE_K_M * height_m
              exponent = -G0_M_S2 / (LAPSE_RATE_K_M * GAS_CONSTANT_AIR_J_KG_K)
              return SEA_LEVEL_PRESSURE_PA * (
                  temperature / SEA_LEVEL_TEMPERATURE_K
              ) ** exponent

          pressure_at_tropopause = SEA_LEVEL_PRESSURE_PA * (
              TROPOPAUSE_TEMPERATURE_K / SEA_LEVEL_TEMPERATURE_K
          ) ** (-G0_M_S2 / (LAPSE_RATE_K_M * GAS_CONSTANT_AIR_J_KG_K))
          return pressure_at_tropopause * math.exp(
              -G0_M_S2
              * (height_m - TROPOPAUSE_HEIGHT_M)
              / (GAS_CONSTANT_AIR_J_KG_K * TROPOPAUSE_TEMPERATURE_K)
          )


      def atmosphere(pressure_altitude_ft: float, sat_c: float) -> AtmosphereState:
          """Combine pressure altitude with measured/modelled static temperature.

          Pressure is fixed by pressure altitude.  Density and sound speed use SAT.
          This avoids treating a temperature deviation as a geometric-altitude shift.
          """

          if not (math.isfinite(pressure_altitude_ft) and math.isfinite(sat_c)):
              raise ValueError("altitude and SAT must be finite")
          temperature_k = sat_c + 273.15
          if temperature_k <= 0:
              raise ValueError("SAT is below absolute zero")
          pressure_pa = pressure_from_pressure_altitude_pa(pressure_altitude_ft)
          density = pressure_pa / (GAS_CONSTANT_AIR_J_KG_K * temperature_k)
          sound_speed = math.sqrt(
              HEAT_CAPACITY_RATIO * GAS_CONSTANT_AIR_J_KG_K * temperature_k
          )
          return AtmosphereState(
              pressure_pa=pressure_pa,
              density_kg_m3=density,
              temperature_k=temperature_k,
              isa_temperature_k=isa_temperature_k(pressure_altitude_ft),
              speed_of_sound_m_s=sound_speed,
          )


      def true_airspeed_m_s(sample: FlightSample) -> float:
          sample.validate()
          return sample.mach * atmosphere(sample.altitude_ft, sample.sat_c).speed_of_sound_m_s


      def _linear_interpolate_reference_fuel_flow(
          thrust_fraction: float,
          parameters: ModelParameters,
      ) -> tuple[float, float]:
          """Interpolate installation-adjusted sea-level reference fuel flow."""

          points = parameters.lto_points
          bounded = min(max(thrust_fraction, points[0][0]), points[-1][0])
          for left, right in zip(points, points[1:]):
              left_power, left_flow, left_installation = left
              right_power, right_flow, right_installation = right
              if bounded <= right_power:
                  fraction = (bounded - left_power) / (right_power - left_power)
                  left_adjusted = left_flow * left_installation
                  right_adjusted = right_flow * right_installation
                  flow = left_adjusted + fraction * (right_adjusted - left_adjusted)
                  return flow, bounded
          power, flow, installation = points[-1]
          return flow * installation, power


      def fuel_flow(
          sample: FlightSample,
          aircraft_mass_kg: float,
          *,
          vertical_rate_fpm: float = 0.0,
          acceleration_m_s2: float = 0.0,
          parameters: ModelParameters | None = None,
      ) -> FuelFlowBreakdown:
          """Estimate clean-configuration two-engine fuel flow.

          BFFM2 was developed as an emissions correction method.  Its use here to
          map public sea-level certification points to altitude is therefore a
          conditional approximation, not an asserted aircraft measurement.
          """

          sample.validate()
          parameters = parameters or ModelParameters()
          parameters.validate()
          if not math.isfinite(aircraft_mass_kg) or not 100_000 <= aircraft_mass_kg <= 310_000:
              raise ValueError("aircraft mass outside audit-model domain")
          if not math.isfinite(vertical_rate_fpm) or abs(vertical_rate_fpm) > 20_000:
              raise ValueError("vertical rate outside audit-model domain")
          if not math.isfinite(acceleration_m_s2) or abs(acceleration_m_s2) > 5:
              raise ValueError("acceleration outside audit-model domain")

          ambient = atmosphere(sample.altitude_ft, sample.sat_c)
          tas_m_s = sample.mach * ambient.speed_of_sound_m_s
          vertical_m_s = vertical_rate_fpm * FPM_TO_M_S
          path_angle_rad = math.atan2(vertical_m_s, tas_m_s)
          load_factor = 1.0 / math.cos(math.radians(sample.bank_deg))

          dynamic_pressure_area_n = (
              0.5
              * ambient.density_kg_m3
              * tas_m_s**2
              * parameters.wing_area_m2
          )
          lift_n = aircraft_mass_kg * G0_M_S2 * math.cos(path_angle_rad) * load_factor
          lift_coefficient = lift_n / dynamic_pressure_area_n
          drag_coefficient = (
              parameters.clean_cd0 + parameters.induced_drag_k * lift_coefficient**2
          )
          drag_n = drag_coefficient * dynamic_pressure_area_n
          required_thrust_n = max(
              0.0,
              drag_n
              + aircraft_mass_kg * G0_M_S2 * math.sin(path_angle_rad)
              + aircraft_mass_kg * acceleration_m_s2,
          )
          raw_thrust_fraction = required_thrust_n / (
              parameters.engine_count * parameters.max_static_thrust_per_engine_n
          )
          reference_flow, bounded_thrust_fraction = _linear_interpolate_reference_fuel_flow(
              raw_thrust_fraction,
              parameters,
          )

          pressure_ratio = ambient.pressure_pa / SEA_LEVEL_PRESSURE_PA
          temperature_ratio = ambient.temperature_k / SEA_LEVEL_TEMPERATURE_K
          mach_factor = math.exp(0.2 * sample.mach**2)
          per_engine_flow = (
              reference_flow
              * pressure_ratio
              / (temperature_ratio**3.8 * mach_factor)
              * parameters.fuel_flow_scale
          )
          total_flow = parameters.engine_count * per_engine_flow

          return FuelFlowBreakdown(
              total_fuel_flow_kg_s=total_flow,
              per_engine_fuel_flow_kg_s=per_engine_flow,
              true_airspeed_m_s=tas_m_s,
              true_airspeed_kt=tas_m_s / KT_TO_M_S,
              lift_coefficient=lift_coefficient,
              drag_n=drag_n,
              required_thrust_n=required_thrust_n,
              thrust_fraction=bounded_thrust_fraction,
              reference_fuel_flow_kg_s_per_engine=reference_flow,
              pressure_ratio=pressure_ratio,
              temperature_ratio=temperature_ratio,
              bffm2_mach_factor=mach_factor,
          )


      def interpolate_sample(start: FlightSample, end: FlightSample, fraction: float) -> FlightSample:

          end.validate()
          parameters = parameters or ModelParameters()
          parameters.validate()
          duration_s = (end.epoch_utc - start.epoch_utc).total_seconds()
          if duration_s <= 0:
              raise ValueError("samples must be strictly increasing in time")
          if not math.isfinite(step_s) or step_s <= 0:
              raise ValueError("step_s must be finite and positive")
          if start_fuel_kg < 0 or zero_fuel_mass_kg <= 0:
              raise ValueError("masses must be non-negative, with positive ZFM")

          vertical_rate_fpm = (end.altitude_ft - start.altitude_ft) / (duration_s / 60.0)
          acceleration_m_s2 = (
              true_airspeed_m_s(end) - true_airspeed_m_s(start)
          ) / duration_s

          fuel_kg = start_fuel_kg
          elapsed_s = 0.0
          exhausted = False
          while elapsed_s < duration_s and fuel_kg > 0:
              this_step_s = min(step_s, duration_s - elapsed_s)
              midpoint = (elapsed_s + 0.5 * this_step_s) / duration_s
              sample = interpolate_sample(start, end, midpoint)
              flow = fuel_flow(
                  sample,
                  zero_fuel_mass_kg + fuel_kg,
                  vertical_rate_fpm=vertical_rate_fpm,
                  acceleration_m_s2=acceleration_m_s2,
                  parameters=parameters,
              ).total_fuel_flow_kg_s
              burn_kg = flow * this_step_s
              if burn_kg >= fuel_kg:
                  fuel_kg = 0.0
                  exhausted = True
                  break
              fuel_kg -= burn_kg
              elapsed_s += this_step_s

          return IntegratedInterval(
              start_utc=start.epoch_utc.isoformat().replace("+00:00", "Z"),
              end_utc=end.epoch_utc.isoformat().replace("+00:00", "Z"),
              duration_s=duration_s,
              vertical_rate_fpm=vertical_rate_fpm,
              acceleration_m_s2=acceleration_m_s2,
              start_fuel_kg=start_fuel_kg,
              end_fuel_kg=fuel_kg,
              fuel_burn_kg=start_fuel_kg - fuel_kg,
              exhausted=exhausted,
          )


      def integrate_trajectory(
          samples: Sequence[FlightSample] | Iterable[FlightSample],
          *,
          start_fuel_kg: float,
          zero_fuel_mass_kg: float,
          step_s: float = 1.0,
          parameters: ModelParameters | None = None,
      ) -> list[IntegratedInterval]:
          """Integrate a trajectory without assigning weights to trajectory families."""

          ordered = tuple(samples)
          if len(ordered) < 2:
              raise ValueError("at least two samples are required")
          output: list[IntegratedInterval] = []
          fuel_kg = start_fuel_kg
          for start, end in zip(ordered, ordered[1:]):
              interval = integrate_interval(
                  start,
                  end,
                  start_fuel_kg=fuel_kg,
                  zero_fuel_mass_kg=zero_fuel_mass_kg,
                  step_s=step_s,
                  parameters=parameters,
              )
              output.append(interval)
              fuel_kg = interval.end_fuel_kg
              if interval.exhausted:
                  break
          return output


      def constant_state_endurance_s(
          sample: FlightSample,
          *,
          start_fuel_kg: float,
          zero_fuel_mass_kg: float,
          step_s: float = 10.0,
          maximum_s: float = 12 * 3600,
          parameters: ModelParameters | None = None,
      ) -> float | None:
          """Return a named stress-case endurance, never an event-time calibration."""

          sample.validate()
          parameters = parameters or ModelParameters()
          fuel_kg = start_fuel_kg
          elapsed_s = 0.0
          while fuel_kg > 0 and elapsed_s < maximum_s:
              flow = fuel_flow(
                  sample,
                  zero_fuel_mass_kg + fuel_kg,
                  parameters=parameters,
              ).total_fuel_flow_kg_s
              this_step_s = min(step_s, fuel_kg / flow, maximum_s - elapsed_s)
              fuel_kg -= flow * this_step_s
              elapsed_s += this_step_s
          return elapsed_s if fuel_kg <= 1e-9 else None
