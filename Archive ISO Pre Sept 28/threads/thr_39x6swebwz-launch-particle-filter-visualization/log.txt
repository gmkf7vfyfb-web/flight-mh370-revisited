      +#!/usr/bin/env python3
      +"""Materialize the read-only Particle Filter Lab API as static JSON assets."""
      +
      +from __future__ import annotations
      +
      +import json
      +import sys
      +from pathlib import Path
      +
      +
      +APP_ROOT = Path(__file__).resolve().parents[1]
      +BACKEND_ROOT = APP_ROOT / "backend"
      +OUTPUT_ROOT = APP_ROOT / "web" / "public" / "static-api"
      +
      +sys.path.insert(0, str(BACKEND_ROOT))
      +
      +from particle_lab.library import library_result, list_library  # noqa: E402
      +from particle_lab.scenarios import default_scenario  # noqa: E402
      +
      +
      +def write_json(path: Path, payload: object) -> None:
      +    path.parent.mkdir(parents=True, exist_ok=True)
      +    path.write_text(
      +        json.dumps(payload, ensure_ascii=False, separators=(",", ":"), allow_nan=False) + "\n",
      +        encoding="utf-8",
      +    )
      +
      +
      +def result_filename(identifier: str) -> str:
      +    family, separator, tag = identifier.partition(":")
      +    if separator != ":" or not family or not tag or any(token in tag for token in fe library identifier: {identifier!r}")
      +    return f"{family}--{tag}.json"
      +
      +
      +def main() -> None:
      +    entries = list_library()
      +    write_json(OUTPUT_ROOT / "scenario.json", default_scenario().model_dump(mode="json"))
      +    write_json(OUTPUT_ROOT / "library.json", entries)
      +
      +    for entry in entries:
      +        identifier = entry["id"]
      +        write_json(OUTPUT_ROOT / "results" / result_filename(identifier), library_result(identifier))
      +
      +    total_bytes = sum(path.stat().st_size for path in OUTPUT_ROOT.rglob("*.json"))
      +    print(f"Wrote {len(entries)} frozen result bundles ({total_bytes / 1_000_000:.1f} MB) to {OUTPUT_ROOT}")
      +
      +
      +if __name__ == "__main__":
      +    main()
