+def mix_hist(key):
+    keys = DW["keys"]; Hp = DW["H"][keys.index(key)].astype(float)
+    s = np.load(WEX / "supplement.npz"); sk = list(s["keys"]); Hs = s["H"][sk.index(key)].astype(float)
+    sD = DW["ess"][key.rsplit("/", 1)[0]]["supplement_share"]
+    return (1 - sD) * Hp / Hp.sum() + (sD * Hs / Hs.sum() if Hs.sum() > 0 else 0)
