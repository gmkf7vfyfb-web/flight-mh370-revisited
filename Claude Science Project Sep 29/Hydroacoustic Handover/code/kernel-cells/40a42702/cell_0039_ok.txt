+def source_filter(x, fs, tau):
+    b, a = signal.butter(1, 1.0 / tau, btype="low", fs=fs)
+    return signal.lfilter(b, a, x)

+
+def pulse(T, r_km, tau, fs=R.FS):
+    h = T[f"h2_{r_km:.1f}"]; fs0 = float(T["fs"])
+    e = np.convolve(h ** 2, np.ones(int(fs0)) / fs0, "same"); pk = int(np.argmax(e))
+    h = np.roll(h, int(R.PULSE_PEAK_S * fs0) - pk)[: int(R.PULSE_LEN_S * fs0)]
+    x = source_filter(h, fs0, tau)
+    sos = signal.butter(4, list(R.BAND), btype="bandpass", fs=fs0, output="sos")
+    x = signal.sosfiltfilt(sos, x)
+    x = signal.resample_poly(x, int(fs), int(fs0))
+    return (x / np.sqrt(np.sum(x ** 2))).astype(float)

+def bank_for(T, ranges, taus, fs=R.FS):
+    bw, be, keys = [], [], []
+    for r in ranges:
+        for tau in taus:
+            p = pulse(T, r, tau, fs); bw.append(p); keys.append((r, tau))
+            e = M.envelope(p, fs); e = e - e.mean(); be.append(e / np.sqrt(np.sum(e ** 2)))
+    return dict(bank_wave=bw, bank_env=be, bank_keys=keys, bank_peak_idx=int(R.PULSE_PEAK_S * fs), bank_env_peak_idx=int(R.PULSE_PEAK_S * 10))

+def peak_level_db(p, fs=R.FS):
+    """max over 1 s frames of the 5-40 Hz level (dB re unit^2) of a pulse; and the pulse energy / peak-1s-energy = T_eff."""
+    _, (lev,) = M.band_levels(np.concatenate([p, np.zeros(fs)]), fs, [R.BAND])

+    e_tot = float(np.sum(p ** 2)) / fs                      # unit^2 s (time-domain energy)
+    return pk, e_tot / (10 ** (pk / 10))

+def wilson(k, n, z=1.96):
+    p = k / n; d = 1 + z * z / n; c = (p + z * z / (2 * n)) / d; h = z * np.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
+    return c - h, c + h

+def fit_logistic(lv, k, n):
+    from scipy.optimize import minimize
+    def nll(th):
+        p = 1 / (1 + np.exp(-(lv - th[0]) / max(th[1], 0.05))); p = np.clip(p, 1e-9, 1 - 1e-9)
+        return -np.sum(k * np.log(p) + (n - k) * np.log(1 - p))
+    th = minimize(nll, [0.0, 3.0], method="Nelder-Mead").x
+    return float(th[0]), float(max(th[1], 0.05))
