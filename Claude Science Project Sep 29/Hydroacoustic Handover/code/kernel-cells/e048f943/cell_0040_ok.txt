+def frac_vs_eta(hist, etas, thr):
+    cdf = np.cumsum(hist); tot = cdf[-1]
+    if tot <= 0: return np.zeros(len(etas))
+    edges = RL.SNR_GRID
+    out = []
+    for eta in etas:
+        x = thr - 10 * np.log10(eta)                     # need snr1 >= x
+        i = np.searchsorted(edges, x, side="left")        # bins with left edge >= x fully count; approximate at 0.1 dB
+        out.append(float(hist[max(i - 1, 0):].sum()))
+    return np.array(out)
