+from __future__ import annotations
+import json, pathlib, sys
+import numpy as np, pandas as pd

+HERE = pathlib.Path(__file__).resolve().parent
+sys.path.insert(0, str(HERE))
+import prereg as R  # noqa: E402
+
+DIG = HERE.parent.parent / "results-data" / "digitised" / "metz2023"
+ELEMENTS = ["H11N1", "H11N2", "H11N3", "H11S1", "H11S2", "H11S3"]
+DT = 0.05
+
+
+def envelope(el):
+    g = pd.read_csv(DIG / "fig3CD_rms_envelopes_0p05s_DIGITISED.csv")
+    g = g[g.element == el].sort_values("utc_s_of_day")
+    return g.utc_s_of_day.values, g.rms_db_re_1uPa.values
+
+
+def windows(t, L):
+    ipk = int(np.argmax(L)); tpk = t[ipk]
+    out = (t < tpk - R.NOISE_EXCL_S) | (t > tpk + R.NOISE_EXCL_S)
+    Ln = float(np.median(L[out]))
+    a = ipk
+    while a > 0 and L[a - 1] > Ln + R.ARR_DB:
+        a -= 1
+    b = ipk
+    while b < len(L) - 1 and L[b + 1] > Ln + R.ARR_DB:
+        b += 1
+    return ipk, Ln, a, b
+
+
+def width(t, L, ipk, drop):
+    thr = L[ipk] - drop; a = ipk; b = ipk
+    while a > 0 and L[a - 1] >= thr:
+        a -= 1
+    while b < len(L) - 1 and L[b + 1] >= thr:
+        b += 1
+    return float(t[b] - t[a] + DT)
+
+
+def element_summary(el):
+    t, L = envelope(el); ipk, Ln, a, b = windows(t, L)
+    ex = np.clip(10 ** (L / 10) - 10 ** (Ln / 10), 0, None)              # uPa^2
+    w = np.zeros_like(ex); w[a: b + 1] = ex[a: b + 1]
+    se = float(w.sum() * DT * 1e-12)                                       # Pa^2 s
+    c = np.cumsum(w) / max(w.sum(), 1e-30)
+    t05, t95 = float(np.interp(0.05, c, t)), float(np.interp(0.95, c, t))
+    Lex = 10 * np.log10(np.maximum(ex, 1e-30))
+    return dict(element=el, triad=el[:4], peak_db=float(L[ipk]), peak_utc_s=float(t[ipk]), noise_db=Ln, snr_peak_db=float(L[ipk] - Ln),
+                arr_start_s=float(t[a]), arr_end_s=float(t[b]), arr_len_s=float(t[b] - t[a] + DT), se_pa2s=se, sel_db=float(10 * np.log10(se / 1e-12)),
+                p_rms_peak_pa=float(10 ** (L[ipk] / 20) * 1e-6), w3_s=width(t, L, ipk, 3), w6_s=width(t, L, ipk, 6), w10_s=width(t, L, ipk, 10),
+                span_5_95_s=t95 - t05, t05_s=t05, t95_s=t95, excess_peak_db=float(Lex[ipk]))
+
+
+def spectrogram(name):
+    z = np.load(DIG / f"fig3AB_spectrogram_{name}_DIGITISED.npz")
+    return z["psd_db_re_1uPa2_Hz"].astype(float), z["utc_s_of_day"], z["freq_hz"]
+
+
+def band_edges(fc):
+    return fc / 2 ** (1 / 6), fc * 2 ** (1 / 6)
+
+
+def band_excess(name, floor_db=80.0, bands=None, t_window=None):
+    """Noise-subtracted band energies (Pa^2 s, absolute through the published PSD scale) over the arrival window of the
+    element's envelope. Returns DataFrame per band and the 4-64 Hz total."""
+    bands = bands or R.BANDS
+    P, tt, ff = spectrogram(name)
+    P = np.where(np.isfinite(P), P, floor_db)
+    lin = 10 ** (P / 10)                                                   # uPa^2/Hz, rows = freq (high -> low)
+    if t_window is None:
+        t, L = envelope(name); ipk, Ln, a, b = windows(t, L); t_window = (t[a], t[b])
+    inw = (tt >= t_window[0]) & (tt <= t_window[1])
+    noise = np.median(lin[:, ~inw], axis=1)                                # per frequency row
+    exc = np.clip(lin[:, inw] - noise[:, None], 0, None)
+    dt = float(np.median(np.diff(tt))); df = float(abs(np.median(np.diff(ff))))
+    rows = []
+    for fc in bands:
+        lo, hi = band_edges(fc); hi_eff = min(hi, 64.0) if fc == 63.0 else hi
+        m = (ff >= lo) & (ff < hi_eff)
+        e = float(exc[m].sum() * df * dt * 1e-12)
+        arr_mean = float(10 * np.log10(lin[m][:, inw].mean())); noi_mean = float(10 * np.log10(noise[m].mean()))
+        rows.append(dict(element=name, fc_hz=fc, f_lo=lo, f_hi=hi_eff, excess_pa2s=e, arrival_mean_psd_db=arr_mean, noise_mean_psd_db=noi_mean,
+                         arrival_minus_noise_db=arr_mean - noi_mean, n_rows=int(m.sum())))
+    m4 = (ff >= 4.0) & (ff < 64.0)
+    tot = float(exc[m4].sum() * df * dt * 1e-12)
+    return pd.DataFrame(rows), tot, t_window
+
+
+def main(out_dir):
+    out = pathlib.Path(out_dir); out.mkdir(parents=True, exist_ok=True)
+    S = pd.DataFrame([element_summary(e) for e in ELEMENTS]); S.to_csv(out / "obs_envelopes.csv", index=False)
+    tri = {}
+    for tr, g in S.groupby("triad"):
+        se = float(g.se_pa2s.mean())
+        tri[tr] = dict(se_pa2s=se, sel_db=float(10 * np.log10(se / 1e-12)), sel_spread_db=float(10 * np.log10(g.se_pa2s.max() / g.se_pa2s.min())),
+                       peak_db_mean=float(10 * np.log10(np.mean(10 ** (g.peak_db / 10)))), noise_db_mean=float(g.noise_db.mean()),
+                       w3_s=float(g.w3_s.mean()), w6_s=float(g.w6_s.mean()), w10_s=float(g.w10_s.mean()), span_5_95_s=float(g.span_5_95_s.mean()),
+                       peak_utc_s_mean=float(g.peak_utc_s.mean()))
+    B = []; chk = {}
+    for name in ("H11N1", "H11S1"):
+        for fl in (80.0, 70.0):
+            b, tot, tw = band_excess(name, fl); b["floor_db"] = fl; B.append(b)
+            se_env = float(S[S.element == name].se_pa2s.iloc[0])
+            chk[f"{name}|floor{fl:g}"] = dict(spectrogram_4_64_excess_pa2s=tot, envelope_se_pa2s=se_env, ratio_db=float(10 * np.log10(tot / se_env)) if tot > 0 else None, window_utc_s=list(map(float, tw)))
+    pd.concat(B).to_csv(out / "obs_band_excess.csv", index=False)
+    json.dump(dict(triads=tri, spectrogram_vs_envelope=chk, label="DIGITISED Metz et al. 2023 Fig. 3"), open(out / "obs_summary.json", "w"), indent=1)
+    print(S.round(3).to_string()); print(json.dumps(tri, indent=1)); print(json.dumps(chk, indent=1))

+if __name__ == "__main__":
+    main(sys.argv[1])
