+"""Reference measurement for one hydrophone triad (collaboration package v2, 22 Oct 2026; DRAFT for Pete's review).
+
+Input: an .npz with  t0_utc (ISO string), fs (Hz), x (3 x N pressure, Pa), lat, lon (3 element positions, deg),
+and a signal window [w0_s, w1_s] (s from t0) plus a pre-event noise window [n0_s, n1_s].
+Per band (4th-order Butterworth, zero phase; bands 2-5, 5-10, 10-20, 20-40 and 5-40 Hz): peak |p| (Pa), RMS (Pa),
+exposure (Pa^2 s) in the window; noise RMS in the pre-event window; SNR = 20 log10(RMS_window / RMS_noise) (dB).
+Bearing: plane-wave fit to the three inter-element delays from the cross-correlation of the band-passed window
+(parabolic sub-sample peak); back-azimuth (deg from N, towards the source) and apparent speed (m/s).
+
+OPTIONAL DETECTOR (--m1b <templates_envelope.npz>): the module's dispersion-aware envelope matched filter M1b
+(weak-signal trials, pre-registration 78c7e88). The 5-40 Hz series is squared, smoothed over 0.5 s and decimated to 10 Hz;
+each template (50 s envelope of a KRAKEN broadband deep-sound-channel arrival for a range and a source duration tau,
+zero-mean, unit energy) is correlated with the local mean removed; the statistic is the maximum correlation coefficient
+rho over the bank. The series rho(t) is returned per element and for the element mean (t = time of the template's
+energy peak). A threshold must be set on the receiver's OWN background (null of 60 s window maxima, as in the module's
+trials); the package gives no threshold. This detector is a shape test and is robust to phase; its demonstrated gain
+(6-9 dB over STA/LTA at P_D = 0.5) is model-matched and is an upper bound on real data.
+
+numpy + scipy only. Usage:
+  python reference_measure.py <input.npz> <w0_s> <w1_s> <n0_s> <n1_s> [out.csv] [--m1b templates_envelope.npz]
+"""
+import sys
+import numpy as np
+from scipy import signal
+
+BANDS = [(2, 5), (5, 10), (10, 20), (20, 40), (5, 40)]
+R_EARTH = 6371008.8
+ENV_FS = 10.0
+ENV_SMOOTH_S = 0.5
+EPS = 1e-30
+
+
+def enu(lat, lon):
+    lat0, lon0 = np.mean(lat), np.mean(lon)
+    e = np.radians(np.asarray(lon) - lon0) * R_EARTH * np.cos(np.radians(lat0))
+    n = np.radians(np.asarray(lat) - lat0) * R_EARTH
+    return np.c_[e, n]
+
+
+def bandpass(x, fs, lo, hi):
+    sos = signal.butter(4, [lo, min(hi, 0.45 * fs)], btype="bandpass", fs=fs, output="sos")
+    return signal.sosfiltfilt(sos, x, axis=-1)
+
+
+def delay(a, b, fs, max_lag):
+    """Delay of b relative to a (s), positive when b arrives later; and the peak normalised cross-correlation."""
+    c = signal.correlate(b, a, mode="full")
+    lags = signal.correlation_lags(len(b), len(a), mode="full")
+    m = np.abs(lags) <= max_lag * fs
+    c, lags = c[m], lags[m]
+    i = int(np.argmax(c))
+    if 0 < i < len(c) - 1:                       # parabolic refinement
+        d = (c[i - 1] - c[i + 1]) / (2 * (c[i - 1] - 2 * c[i] + c[i + 1]))
+    else:
+        d = 0.0
+    return (lags[i] + d) / fs, float(c[i] / np.sqrt(np.sum(a * a) * np.sum(b * b)))
+
+
+def plane_wave(xy, x, fs):
+    """Slowness s (s/m) from tau_ij = s . (r_j - r_i); back-azimuth points to the source."""
+    pairs = [(0, 1), (0, 2), (1, 2)]
+    max_lag = np.max(np.linalg.norm(xy[:, None] - xy[None], axis=-1)) / 1300.0
+    taus, ccs, G = [], [], []
+    for i, j in pairs:
+        t, cc = delay(x[i], x[j], fs, max_lag)
+        taus.append(t); ccs.append(cc); G.append(xy[j] - xy[i])
+    s, *_ = np.linalg.lstsq(np.array(G), np.array(taus), rcond=None)
+    baz = np.degrees(np.arctan2(-s[0], -s[1])) % 360
+    resid = np.array(taus) - np.array(G) @ s
+    return baz, 1 / np.linalg.norm(s), float(np.min(ccs)), float(np.sqrt(np.mean(resid ** 2)))
+
+
+def envelope(x, fs, smooth_s=ENV_SMOOTH_S, out_fs=ENV_FS):
+    """Squared pressure, boxcar-smoothed over smooth_s, block-averaged to out_fs (the module's M1b front end)."""
+    e = np.asarray(x, float) ** 2
+    k = max(int(smooth_s * fs), 1)
+    e = np.convolve(e, np.ones(k) / k, "same")
+    d = int(round(fs / out_fs))
+    n = len(e) // d
+    return e[: n * d].reshape(n, d).mean(1)
+
+
+def m1b(x_band, fs, templates):
+    """M1b on one 5-40 Hz series. templates: dict with 'env' (K x L, zero-mean unit-energy at 10 Hz), 'peak_idx'.
+    Returns (t_s, rho): rho(t) = max over the bank of the correlation coefficient with the local mean removed,
+    t = time (s from the series start) of the template's energy peak."""
+    e = envelope(x_band, fs)
+    best = None
+    for h in templates["env"]:
+        L = len(h)
+        if len(e) < L:
+            continue
+        c = signal.fftconvolve(e, h[::-1], mode="valid")
+        cs1 = np.concatenate([[0.0], np.cumsum(e)]); cs2 = np.concatenate([[0.0], np.cumsum(e ** 2)])
+        s1 = cs1[L:] - cs1[:-L]; s2 = cs2[L:] - cs2[:-L]
+        var = np.maximum(s2 - s1 ** 2 / L, EPS)
+        r = c / np.sqrt(var * np.sum(h ** 2))
+        best = r if best is None else np.maximum(best[: len(r)], r[: len(best)])
+    if best is None:
+        return np.array([]), np.array([])
+    i = np.arange(len(best))
+    return (i + int(templates["peak_idx"])) / ENV_FS, best
+
+
+def load_templates(path):
+    z = np.load(path, allow_pickle=False)
+    return dict(env=z["env"], peak_idx=int(z["peak_idx"]), ranges_km=z["ranges_km"], taus_s=z["taus_s"], fs=float(z["fs"]))
+
+
+def measure(d, w0, w1, n0, n1, templates=None):
+    fs, x = float(d["fs"]), np.asarray(d["x"], float)
+    xy = enu(d["lat"], d["lon"])
+    iw, inn = slice(int(w0 * fs), int(w1 * fs)), slice(int(n0 * fs), int(n1 * fs))
+    rows = []
+    for lo, hi in BANDS:
+        y = bandpass(x, fs, lo, hi)
+        yw, yn = y[:, iw], y[:, inn]
+        rms, nrms = np.sqrt(np.mean(yw ** 2, 1)), np.sqrt(np.mean(yn ** 2, 1))
+        baz, vapp, ccmin, rres = plane_wave(xy, yw, fs)
+        row = dict(band=f"{lo}-{hi}", peak_pa=float(np.abs(yw).max()), rms_pa=float(rms.mean()),
+                   exposure_pa2s=float(np.mean(np.sum(yw ** 2, 1) / fs)), noise_rms_pa=float(nrms.mean()),
+                   snr_db=float(20 * np.log10(rms.mean() / nrms.mean())), backazimuth_deg=float(baz),
+                   apparent_speed_m_s=float(vapp), min_xcorr=ccmin, delay_resid_s=rres,
+                   m1b_rho_max="", m1b_t_peak_s="", m1b_rho_noise_max="")
+        if templates is not None and (lo, hi) == (5, 40):
+            t, r = m1b(y.mean(0), fs, templates)                 # element mean: coherent gain for a plane wave
+            if len(r):
+                mw = (t >= w0) & (t <= w1); mn = (t >= n0) & (t <= n1)
+                if mw.any():
+                    k = int(np.argmax(np.where(mw, r, -np.inf)))
+                    row["m1b_rho_max"] = float(r[k]); row["m1b_t_peak_s"] = float(t[k])
+                if mn.any():
+                    row["m1b_rho_noise_max"] = float(r[mn].max())
+        rows.append(row)
+    return rows
+
+
+if __name__ == "__main__":
+    import csv
+    args = sys.argv[1:]
+    templates = None
+    if "--m1b" in args:
+        k = args.index("--m1b"); templates = load_templates(args[k + 1]); del args[k:k + 2]
+    d = np.load(args[0], allow_pickle=False)
+    rows = measure(d, *map(float, args[1:5]), templates=templates)
+    out = open(args[5], "w", newline="") if len(args) > 5 else sys.stdout
+    w = csv.DictWriter(out, fieldnames=list(rows[0]))
+    w.writeheader(); w.writerows(rows)
